use bevy::prelude::*;
use bevy_vn::prelude::*;

fn main() {
    App::new()
        .add_plugins(DefaultPlugins)
        .add_plugin(VnEnginePlugin)
        .run();
}

vn_script! {
    // ==========================================
    // 资源定义 (无分号，使用 =)
    // ==========================================
    character n1 {
        name = "未来的约瑟"
        sprite = "images/sidebar/1000.png"
    }

    character n15 {
        name = "精灵王约瑟"
        sprite = "images/sidebar/6.png"
    }

    background "bg forest" {
        image = "images/background/bg4.png"
    }

    // ==========================================
    // 脚本逻辑 (无分号，无箭头)
    // ==========================================

    label start {
        scene "bg forest" with FadeIn
        
        show n15
        
        say n15 "我的友人们啊，此刻即使我竭尽全力避免了毁灭的终末..."

        // 菜单语法：选项文本后直接跟大括号
        menu {
            "询问关于伊的事" {
                // 可以在这里执行多条语句
                say n15 "好的，让我告诉你关于伊的事。"
                jump branch_question_about_yi
            }
            "沉默" {
                say n15 "......你选择沉默吗？"
                jump bs2
            }
        }
    }

    label branch_question_about_yi {
        say n15 "「伊」……向我提起「伊」的那些存在..."
        jump bs2
    }
    
    label bs2 {
        say n1 "了解了。"
        return
    }
}
