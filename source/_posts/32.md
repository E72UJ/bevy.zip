## 过程宏
```rust
use proc_macro::TokenStream;
use quote::quote;
use syn::{
    parse::{Parse, ParseStream, Result},
    Ident, LitStr, Token, braced,
    parse_macro_input,
};

// ==========================================
// 1. 定义语法树结构
// ==========================================

struct VnScript {
    items: Vec<ScriptItem>,
}

enum ScriptItem {
    Character(CharacterDef),
    Background(BackgroundDef),
    Image(ImageDef),
    Label(LabelDef),
}

struct CharacterDef {
    name: Ident,
    props: Vec<Prop>,
}

struct BackgroundDef {
    name: LitStr,
    props: Vec<Prop>,
}

struct ImageDef {
    name: LitStr,
    props: Vec<Prop>,
}

struct Prop {
    key: Ident,
    value: PropertyValue,
}

enum PropertyValue {
    Str(LitStr),
    Ident(Ident),
}

struct LabelDef {
    name: Ident,
    commands: Vec<Command>,
}

enum Command {
    Scene(SceneCommand),
    Show(ShowCommand),
    Say(SayCommand),
    Menu(MenuCommand),
    Jump(JumpCommand),
    Exit, // 改为 Exit
}

struct SceneCommand {
    bg_name: LitStr,
    transition: Option<Ident>,
}

struct ShowCommand {
    char_name: Ident, 
}

struct SayCommand {
    char_name: Ident,
    text: LitStr,
}

struct JumpCommand {
    target: Ident,
}

struct MenuCommand {
    options: Vec<MenuOption>,
}

struct MenuOption {
    text: LitStr,
    commands: Vec<Command>,
}

// ==========================================
// 2. 实现 Syn 解析器
// ==========================================

fn match_ident(input: ParseStream, expected: &str) -> bool {
    let fork = input.fork();
    if let Ok(ident) = fork.parse::<Ident>() {
        if ident == expected {
            return true;
        }
    }
    false
}

impl Parse for VnScript {
    fn parse(input: ParseStream) -> Result<Self> {
        let mut items = Vec::new();
        while !input.is_empty() {
            if match_ident(input, "character") {
                items.push(ScriptItem::Character(input.parse()?));
            } else if match_ident(input, "background") {
                items.push(ScriptItem::Background(input.parse()?));
            } else if match_ident(input, "image") {
                items.push(ScriptItem::Image(input.parse()?));
            } else if match_ident(input, "label") {
                items.push(ScriptItem::Label(input.parse()?));
            } else {
                return Err(input.error("预期 character, background, image 或 label"));
            }
        }
        Ok(VnScript { items })
    }
}

impl Parse for CharacterDef {
    fn parse(input: ParseStream) -> Result<Self> {
        let ident: Ident = input.parse()?; 
        if ident != "character" { return Err(syn::Error::new(ident.span(), "预期 character")); }
        
        let name = input.parse()?;
        let content;
        braced!(content in input);
        let props = content.call(parse_props)?;
        Ok(CharacterDef { name, props })
    }
}

impl Parse for BackgroundDef {
    fn parse(input: ParseStream) -> Result<Self> {
        let ident: Ident = input.parse()?; 
        if ident != "background" { return Err(syn::Error::new(ident.span(), "预期 background")); }
        
        let name = input.parse()?;
        let content;
        braced!(content in input);
        let props = content.call(parse_props)?;
        Ok(BackgroundDef { name, props })
    }
}

impl Parse for ImageDef {
    fn parse(input: ParseStream) -> Result<Self> {
        let ident: Ident = input.parse()?; 
        if ident != "image" { return Err(syn::Error::new(ident.span(), "预期 image")); }
        
        let name = input.parse()?;
        let content;
        braced!(content in input);
        let props = content.call(parse_props)?;
        Ok(ImageDef { name, props })
    }
}

impl Parse for LabelDef {
    fn parse(input: ParseStream) -> Result<Self> {
        let ident: Ident = input.parse()?; 
        if ident != "label" { return Err(syn::Error::new(ident.span(), "预期 label")); }
        
        let name = input.parse()?;
        let content;
        braced!(content in input);
        let commands = content.call(parse_commands)?;
        Ok(LabelDef { name, commands })
    }
}

fn parse_props(input: ParseStream) -> Result<Vec<Prop>> {
    let mut props = Vec::new();
    while !input.is_empty() {
        let key: Ident = input.parse()?;
        input.parse::<Token![=]>()?;
        
        let lookahead = input.lookahead1();
        let value = if lookahead.peek(LitStr) {
            PropertyValue::Str(input.parse()?)
        } else if lookahead.peek(Ident) {
            PropertyValue::Ident(input.parse()?)
        } else {
            return Err(lookahead.error());
        };

        props.push(Prop { key, value });
        let _ = input.parse::<Token![,]>();
    }
    Ok(props)
}

fn parse_commands(input: ParseStream) -> Result<Vec<Command>> {
    let mut commands = Vec::new();
    while !input.is_empty() {
        let cmd = parse_command(input)?;
        commands.push(cmd);
        let _ = input.parse::<Token![;]>();
    }
    Ok(commands)
}

fn parse_command(input: ParseStream) -> Result<Command> {
    // 现在所有指令都是普通标识符，统一解析
    let ident: Ident = input.parse()?;
    let ident_str = ident.to_string();

    match ident_str.as_str() {
        "scene" => {
            let bg_name = input.parse()?;
            let transition = if match_ident(input, "with") {
                input.parse::<Ident>()?; // 消耗 with
                Some(input.parse()?)
            } else {
                None
            };
            Ok(Command::Scene(SceneCommand { bg_name, transition }))
        },
        "show" => {
            let char_name = input.parse()?;
            Ok(Command::Show(ShowCommand { char_name }))
        },
        "say" => {
            let char_name = input.parse()?;
            let text = input.parse()?;
            Ok(Command::Say(SayCommand { char_name, text }))
        },
        "menu" => {
            let content;
            braced!(content in input);
            let mut options = Vec::new();
            
            while !content.is_empty() {
                let text = content.parse()?;
                let inner;
                braced!(inner in content);
                let cmds = inner.call(parse_commands)?;
                options.push(MenuOption { text, commands: cmds });
                let _ = content.parse::<Token![,]>();
            }
            Ok(Command::Menu(MenuCommand { options }))
        },
        "jump" => {
            let target = input.parse()?;
            Ok(Command::Jump(JumpCommand { target }))
        },
        "exit" => {
            // 解析 exit 指令
            Ok(Command::Exit)
        },
        _ => Err(syn::Error::new(ident.span(), format!("未知指令: {}", ident_str)))
    }
}

// 3. 代码生成
// ==========================================

impl VnScript {
    fn to_tokens(&self) -> proc_macro2::TokenStream {
        let mut resource_tokens = Vec::new();
        let mut label_tokens = Vec::new();

        for item in &self.items {
            match item {
                ScriptItem::Character(c) => resource_tokens.push(c.to_tokens()),
                ScriptItem::Background(b) => resource_tokens.push(b.to_tokens()),
                ScriptItem::Image(i) => resource_tokens.push(i.to_tokens()),
                ScriptItem::Label(l) => label_tokens.push(l.to_tokens()),
            }
        }

        quote! {
            fn run_vn_script() {
                println!("=== 资源初始化 ===");
                #(#resource_tokens)*
                
                println!("\n=== 脚本开始执行 ===");
                let mut current_label = "start".to_string();
                
                'script_loop: loop {
                    match current_label.as_str() {
                        #(#label_tokens)*
                        _ => {
                            println!("错误: 未知的 Label '{}'", current_label);
                            break 'script_loop;
                        }
                    }
                }
            }
        }
    }
}

impl CharacterDef {
    fn to_tokens(&self) -> proc_macro2::TokenStream {
        let name = &self.name;
        let mut char_name_str = "Unknown".to_string();
        
        for p in &self.props {
            if p.key == "name" {
                if let PropertyValue::Str(s) = &p.value {
                    char_name_str = s.value();
                }
            }
        }

        quote! {
            println!("定义角色: {} (显示名称: {})", stringify!(#name), #char_name_str);
        }
    }
}

impl BackgroundDef {
    fn to_tokens(&self) -> proc_macro2::TokenStream {
        let name = &self.name;
        quote! {
            println!("定义背景: {} ", #name);
        }
    }
}

impl ImageDef {
    fn to_tokens(&self) -> proc_macro2::TokenStream {
        let name = &self.name;
        quote! {
            println!("定义图像: {} ", #name);
        }
    }
}

impl LabelDef {
    fn to_tokens(&self) -> proc_macro2::TokenStream {
        let name_str = self.name.to_string();
        let cmd_tokens: Vec<_> = self.commands.iter().map(|c| c.to_tokens()).collect();

        quote! {
            #name_str => {
                println!("\n[进入 Label: {}]", #name_str);
                #(#cmd_tokens)*
            }
        }
    }
}

impl Command {
    fn to_tokens(&self) -> proc_macro2::TokenStream {
        match self {
            Command::Scene(cmd) => {
                let bg = &cmd.bg_name;
                let trans = if let Some(t) = &cmd.transition {
                    quote!(println!("  (转场特效: {})", stringify!(#t));)
                } else {
                    quote!()
                };
                quote! {
                    println!("指令: 显示背景 {}", #bg);
                    #trans
                }
            }
            Command::Show(cmd) => {
                let char_name = &cmd.char_name;
                quote! {
                    println!("指令: 显示立绘 {}", stringify!(#char_name));
                }
            }
            Command::Say(cmd) => {
                let char_name = &cmd.char_name;
                let text = &cmd.text;
                quote! {
                    println!("对话 [{}]: {}", stringify!(#char_name), #text);
                }
            }
            Command::Menu(menu) => {
                let mut option_matchers = Vec::new();
                
                for (idx, opt) in menu.options.iter().enumerate() {
                    let text = &opt.text;
                    let inner_cmds: Vec<_> = opt.commands.iter().map(|c| c.to_tokens()).collect();
                    let idx = idx + 1;
                    
                    option_matchers.push(quote! {
                        #idx => {
                            println!("用户选择: {}", #text);
                            #(#inner_cmds)*
                        }
                    });
                }

                quote! {
                    println!("指令: 显示菜单");
                    let choice = 1; 
                    match choice {
                        #(#option_matchers)*
                        _ => {}
                    }
                }
            }
            Command::Jump(cmd) => {
                let target_str = cmd.target.to_string();
                quote! {
                    println!("指令: 跳转 -> {}", #target_str);
                    current_label = #target_str.to_string();
                    continue 'script_loop;
                }
            }
            Command::Exit => {
                quote! {
                    println!("指令: 退出脚本");
                    break 'script_loop;
                }
            }
        }
    }
}

// ==========================================
// 4. 宏入口
// ==========================================

#[proc_macro]
pub fn vn_script(input: TokenStream) -> TokenStream {
    let script = parse_macro_input!(input as VnScript);
    script.to_tokens().into()
}
```
