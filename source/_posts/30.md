#[derive(Component)]
struct TypewriterText {
    full_text: String,
    current_index: usize,
    timer: Timer,
    is_complete: bool,
}

impl TypewriterText {
    fn new(text: String, chars_per_second: f32) -> Self {
        let duration = 1.0 / chars_per_second;
        Self {
            full_text: text,
            current_index: 0,
            timer: Timer::from_seconds(duration, TimerMode::Repeating),
            is_complete: false,
        }
    }

    fn get_current_text(&self) -> String {
        self.full_text.chars().take(self.current_index).collect()
    }

    fn skip_to_end(&mut self) {
        self.current_index = self.full_text.chars().count();
        self.is_complete = true;
    }
}

fn update_typewriter_text(
    time: Res<Time>,
    mut query: Query<(&mut TypewriterText, &mut Text), With<DialogueText>>,
    mut dialogue_system: ResMut<DialogueSystem>,
    font_assets: Res<FontAssets>,
    image_assets: Res<ImageAssets>, // 修改：使用 ImageAssets
    choice_container_query: Query<Entity, With<ChoiceContainer>>,
    mut commands: Commands,
) {
    if dialogue_system.is_showing_choices {
        return;
    }

    for (mut typewriter, mut text) in query.iter_mut() {
        if !typewriter.is_complete {
            typewriter.timer.tick(time.delta());

            if typewriter.timer.just_finished() {
                typewriter.current_index += 1;

                let char_count = typewriter.full_text.chars().count();
                if typewriter.current_index >= char_count {
                    typewriter.is_complete = true;
                    dialogue_system.is_waiting_for_input = true;
                }

                **text = typewriter.get_current_text();
            }
        } else if typewriter.is_complete && !dialogue_system.choices_shown {
            let choices_data: Option<Vec<Choice>> = {
                let index = dialogue_system.current_dialogue_index;
                if let Some(dialogue_data) = dialogue_system.dialogues.get(index) {
                    if let DialogueNode::Choice { choices, .. } = &dialogue_data.node {
                        Some(choices.clone())
                    } else {
                        None
                    }
                } else {
                    None
                }
            };

            if let Some(choices) = choices_data {
                dialogue_system.is_showing_choices = true;
                dialogue_system.choices_shown = true;
                
                if let Ok(container_entity) = choice_container_query.single() {
                    // 修改：传递 image_assets
                    show_choices_ui(&mut commands, &font_assets, &image_assets, container_entity, &choices);
                }
            }
        }
    }
}

fn show_choices_ui(
    commands: &mut Commands,
    font_assets: &Res<FontAssets>,
    image_assets: &Res<ImageAssets>,
    container_entity: Entity,
    choices: &[Choice],
) {
    commands.entity(container_entity).with_children(|parent| {
        for (i, choice) in choices.iter().enumerate() {
            parent
                .spawn((
                    ChoiceButton,
                    ChoiceButtonIndex(i),
                    Button,
                    Node {
                        width: Val::Px(467.0), // 修改：宽度 467
                        height: Val::Px(95.0), // 修改：高度 35
                        justify_content: JustifyContent::Center,
                        align_items: AlignItems::Center,
                        margin: UiRect::all(Val::Px(5.0)),
                        ..default()
                    },
                    ImageNode::new(image_assets.choice_button_normal.clone()),
                ))
                .with_children(|parent| {
                    parent.spawn((
                        Text::new(&choice.text),
                        TextColor(Color::WHITE),

                            font: font_assets.main.clone(),

                            ..default()
                        },
                    ));
                });
        }
    });
    
    println!("显示 {} 个选项", choices.len());
}

// 新增：处理选项按钮的悬停效果
fn handle_choice_hover(
    image_assets: Res<ImageAssets>,
    mut query: Query<(&Interaction, &mut ImageNode), (Changed<Interaction>, With<ChoiceButton>)>,
) {
    for (interaction, mut image_node) in query.iter_mut() {
        match interaction {
            Interaction::Hovered | Interaction::Pressed => {
                image_node.image = image_assets.choice_button_hover.clone();
            }
            Interaction::None => {
                image_node.image = image_assets.choice_button_normal.clone();
            }
        }
    }
}

fn handle_autoplay_toggle(
    keyboard_input: Res<ButtonInput<KeyCode>>,
    mut autoplay: ResMut<AutoPlay>,
    mut indicator_query: Query<&mut Text, With<AutoPlayIndicator>>,
    dialogue_system: Res<DialogueSystem>,
) {
    if dialogue_system.is_showing_choices {
        return;
    }

    if keyboard_input.just_pressed(KeyCode::KeyA) {
        autoplay.enabled = !autoplay.enabled;
        autoplay.timer = Timer::from_seconds(autoplay.wait_duration, TimerMode::Once);
        
        for mut text in indicator_query.iter_mut() {
            **text = if autoplay.enabled {
                "自动播放: 开启".to_string()
            } else {
                "自动播放: 关闭".to_string()
            };
        }
    }
}

fn update_autoplay(
    time: Res<Time>,
    mut autoplay: ResMut<AutoPlay>,
    mut dialogue_system: ResMut<DialogueSystem>,
    style_map: Res<CharacterStyleMap>,
    mut namebox_query: Query<&mut Node, With<Namebox>>,
    mut speaker_query: Query<&mut Text, (With<SpeakerNameText>, Without<DialogueText>)>,
    mut dialogue_query: Query<(&mut Text, &mut TypewriterText), With<DialogueText>>,
    font_assets: Res<FontAssets>,
    choice_container_query: Query<Entity, With<ChoiceContainer>>,
    mut commands: Commands,
) {
    if !autoplay.enabled || dialogue_system.is_showing_choices {
        return;
    }

    let typewriter_complete = dialogue_query.iter().all(|(_, tw)| tw.is_complete);
    
    if !typewriter_complete {
        return;
    }

    autoplay.timer.tick(time.delta());

    if autoplay.timer.just_finished() {
        advance_dialogue(
            &mut dialogue_system,
            &style_map,
            &mut namebox_query,
            &mut speaker_query,
            &mut dialogue_query,
            &font_assets,
            &mut commands,
            &choice_container_query,
        );
        
        autoplay.timer = Timer::from_seconds(autoplay.wait_duration, TimerMode::Once);
    }
}

fn handle_click_input(
    mut interaction_query: Query<&Interaction, (Changed<Interaction>, With<ClickArea>)>,
    mut dialogue_system: ResMut<DialogueSystem>,
    mut autoplay: ResMut<AutoPlay>,
    style_map: Res<CharacterStyleMap>,
    mut namebox_query: Query<&mut Node, With<Namebox>>,
    mut speaker_query: Query<&mut Text, (With<SpeakerNameText>, Without<DialogueText>)>,
    mut dialogue_query: Query<(&mut Text, &mut TypewriterText), With<DialogueText>>,
    font_assets: Res<FontAssets>,
    choice_container_query: Query<Entity, With<ChoiceContainer>>,
    mut commands: Commands,
) {
    if dialogue_system.is_showing_choices {
        return;
    }

    for interaction in &mut interaction_query {
        if let Interaction::Pressed = *interaction {
            autoplay.timer = Timer::from_seconds(autoplay.wait_duration, TimerMode::Once);
            
            let mut any_typing = false;

            if let Ok((mut dialogue_text, mut typewriter)) = dialogue_query.single_mut() {
                if !typewriter.is_complete {
                    typewriter.skip_to_end();
                    **dialogue_text = typewriter.get_current_text();
                    dialogue_system.is_waiting_for_input = true;
                    any_typing = true;
                }
            }

            if !any_typing && dialogue_system.is_waiting_for_input {
                advance_dialogue(
                    &mut dialogue_system,
                    &style_map,
                    &mut namebox_query,
                    &mut speaker_query,
                    &mut dialogue_query,
                    &font_assets,
                    &mut commands,
                    &choice_container_query,
                );
            }
        }
    }
}

fn handle_dialogue_input(
    keyboard_input: Res<ButtonInput<KeyCode>>,
    mut dialogue_system: ResMut<DialogueSystem>,
    mut autoplay: ResMut<AutoPlay>,
    style_map: Res<CharacterStyleMap>,
    mut namebox_query: Query<&mut Node, With<Namebox>>,
    mut speaker_query: Query<&mut Text, (With<SpeakerNameText>, Without<DialogueText>)>,
    mut dialogue_query: Query<(&mut Text, &mut TypewriterText), With<DialogueText>>,
    font_assets: Res<FontAssets>,
    choice_container_query: Query<Entity, With<ChoiceContainer>>,
    mut commands: Commands,
) {
    if dialogue_system.is_showing_choices {
        return;
    }

    if keyboard_input.just_pressed(KeyCode::Space) {
        autoplay.timer = Timer::from_seconds(autoplay.wait_duration, TimerMode::Once);
        
        let mut any_typing = false;

        if let Ok((mut dialogue_text, mut typewriter)) = dialogue_query.single_mut() {
            if !typewriter.is_complete {
                typewriter.skip_to_end();
                **dialogue_text = typewriter.get_current_text();
                dialogue_system.is_waiting_for_input = true;
                any_typing = true;
            }
        }

        if !any_typing && dialogue_system.is_waiting_for_input {
            advance_dialogue(
                &mut dialogue_system,
                &style_map,
                &mut namebox_query,
                &mut speaker_query,
                &mut dialogue_query,
                &font_assets,
                &mut commands,
                &choice_container_query,
            );
        }
    }
}

fn handle_choice_input(
    mut interaction_query: Query<(&Interaction, &ChoiceButtonIndex), (Changed<Interaction>, With<ChoiceButton>)>,
    mut dialogue_system: ResMut<DialogueSystem>,
    style_map: Res<CharacterStyleMap>,
    mut namebox_query: Query<&mut Node, With<Namebox>>,
    mut speaker_query: Query<&mut Text, (With<SpeakerNameText>, Without<DialogueText>)>,
    mut dialogue_query: Query<(&mut Text, &mut TypewriterText), With<DialogueText>>,
    // 参数类型：查询 Children 组件
    choice_container_query: Query<&Children, With<ChoiceContainer>>,
    mut commands: Commands,
) {
    for (interaction, index) in &mut interaction_query {
        if let Interaction::Pressed = *interaction {
            let jump_target = get_choice_jump_target(&dialogue_system, index.0);
            
            if let Some(target) = jump_target {
                // 获取容器内的子元素
                if let Ok(children) = choice_container_query.single() {
                    // 修正：child 已经是 Entity 类型，不需要 * 解引用
                    // 使用 despawn_recursive 确保连同按钮上的文字一起销毁
                    for child in children.iter() {
                        commands.entity(child).despawn();
                    }
                }

                dialogue_system.is_showing_choices = false;
                dialogue_system.choices_shown = false;

                jump_to_dialogue(
                    target,
                    &mut dialogue_system,
                    &style_map,
                    &mut namebox_query,
                    &mut speaker_query,
                    &mut dialogue_query,
                );
            }
        }
    }
}

fn get_choice_jump_target(dialogue_system: &DialogueSystem, choice_index: usize) -> Option<usize> {
    if let Some(dialogue_data) = dialogue_system.dialogues.get(dialogue_system.current_dialogue_index) {
        if let DialogueNode::Choice { choices, .. } = &dialogue_data.node {
            return choices.get(choice_index).map(|c| c.jump_to);
        }
    }
    None
}

fn jump_to_dialogue(
    target: usize,
    dialogue_system: &mut ResMut<DialogueSystem>,
    style_map: &Res<CharacterStyleMap>,
    namebox_query: &mut Query<&mut Node, With<Namebox>>,
    speaker_query: &mut Query<&mut Text, (With<SpeakerNameText>, Without<DialogueText>)>,
    dialogue_query: &mut Query<(&mut Text, &mut TypewriterText), With<DialogueText>>,
) {
    if target >= dialogue_system.dialogues.len() {
        println!("跳转目标超出范围: {}", target);
        return;
    }

    let (speaker, text, _) = get_dialogue_info(&dialogue_system.dialogues[target]);

    dialogue_system.current_dialogue_index = target;
    dialogue_system.is_waiting_for_input = false;

    if let Ok(mut speaker_text) = speaker_query.single_mut() {
        **speaker_text = speaker.clone();
    }

    if let Ok((mut dialogue_text, mut typewriter)) = dialogue_query.single_mut() {
        typewriter.full_text = text;
        typewriter.current_index = 0;
        typewriter.is_complete = false;
        **dialogue_text = String::new();
    }

    if let Ok(mut namebox_node) = namebox_query.single_mut() {
        if let Some(style) = style_map.map.get(&speaker) {
            namebox_node.left = style.left;
            namebox_node.bottom = style.bottom;
        } else {
            namebox_node.left = Val::Px(550.0);
            namebox_node.bottom = Val::Px(400.0);
        }
    }

    println!("跳转到对话 {}", target);
}

fn advance_dialogue(
    dialogue_system: &mut ResMut<DialogueSystem>,
    style_map: &Res<CharacterStyleMap>,
    namebox_query: &mut Query<&mut Node, With<Namebox>>,
    speaker_query: &mut Query<&mut Text, (With<SpeakerNameText>, Without<DialogueText>)>,
    dialogue_query: &mut Query<(&mut Text, &mut TypewriterText), With<DialogueText>>,
    _font_assets: &Res<FontAssets>,
    _commands: &mut Commands,
    _choice_container_query: &Query<Entity, With<ChoiceContainer>>,
) {
    let next_index = dialogue_system.current_dialogue_index + 1;

    if next_index < dialogue_system.dialogues.len() {
        let (speaker, text, _) = get_dialogue_info(&dialogue_system.dialogues[next_index]);

        dialogue_system.current_dialogue_index = next_index;
        dialogue_system.is_waiting_for_input = false;
        dialogue_system.choices_shown = false;

        if let Ok(mut speaker_text) = speaker_query.single_mut() {
            **speaker_text = speaker.clone();
        }

        if let Ok((mut dialogue_text, mut typewriter)) = dialogue_query.single_mut() {
            typewriter.full_text = text;
            typewriter.current_index = 0;
            typewriter.is_complete = false;
            **dialogue_text = String::new();
        }

        if let Ok(mut namebox_node) = namebox_query.single_mut() {
            if let Some(style) = style_map.map.get(&speaker) {
                namebox_node.left = style.left;
                namebox_node.bottom = style.bottom;
            } else {
                namebox_node.left = Val::Px(550.0);
                namebox_node.bottom = Val::Px(400.0);
            }
        }
    } else {
        let (speaker, text, _) = get_dialogue_info(&dialogue_system.dialogues[0]);

        dialogue_system.current_dialogue_index = 0;
        dialogue_system.is_waiting_for_input = false;
        dialogue_system.choices_shown = false;

        if let Ok(mut speaker_text) = speaker_query.single_mut() {
            **speaker_text = speaker.clone();
        }

        if let Ok((mut dialogue_text, mut typewriter)) = dialogue_query.single_mut() {
            typewriter.full_text = text;
            typewriter.current_index = 0;
            typewriter.is_complete = false;
            **dialogue_text = String::new();
        }

        if let Ok(mut namebox_node) = namebox_query.single_mut() {
            if let Some(style) = style_map.map.get(&speaker) {
                namebox_node.left = style.left;
                namebox_node.bottom = style.bottom;
            } else {
                namebox_node.left = Val::Px(550.0);
                namebox_node.bottom = Val::Px(400.0);
            }
        }

        println!("对话循环重新开始");
    }
}

fn handle_playing_input(
    keyboard_input: Res<ButtonInput<KeyCode>>,
    mut next_state: ResMut<NextState<GameState>>,
) {
    if keyboard_input.just_pressed(KeyCode::Escape) {
        next_state.set(GameState::Menu);
    }
}

fn cleanup_playing(
    mut commands: Commands,
    query: Query<Entity, With<PlayingEntity>>,
    mut autoplay: ResMut<AutoPlay>,
) {
    for entity in query.iter() {
        commands.entity(entity).despawn();
    }
    
    autoplay.enabled = false;
    println!("退出游戏");
}

fn update_ui_scale(
    windows: Query<&Window>,
    mut ui_scale: ResMut<UiScale>,
) {
    if let Some(window) = windows.iter().next() {
        let scale = (window.width() / VIRTUAL_WIDTH).min(window.height() / VIRTUAL_HEIGHT);
        ui_scale.0 = scale;
    }
}

fn update_character_image(
    dialogue_system: Res<DialogueSystem>,
    asset_server: Res<AssetServer>,
    mut character_query: Query<&mut ImageNode, With<CharacterImage>>,
) {
    if let Ok(mut image_node) = character_query.single_mut() {
        let (_, _, character_image) = get_dialogue_info(&dialogue_system.dialogues[dialogue_system.current_dialogue_index]);
        image_node.image = asset_server.load(&character_image);
    }
}
